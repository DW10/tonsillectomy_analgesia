make_cohort <- function(){
rest_of_chapter <- c(paste0("F0", 0:9), paste0("F", 10:33),  paste0("F", 36:99))

q1 <- cohort_encounter_selection(cohort_name = "Tonsillectomy",
                          include_OPCS =  c("F34", "F35"),
                          exclude_OPCS = c("E20",rest_of_chapter),
                          min_age = NA,
                          max_age = NA,
                          case_priority = "elective",
                          return_index_procedures_only=TRUE,
                          verbose = TRUE
                          )

q2 <- cohort_encounter_selection(cohort_name = "Adenoidectomy",
                          include_OPCS =  c("E20"),
                          exclude_OPCS = c("F34", "F35",rest_of_chapter),
                          min_age = NA,
                          max_age = NA,
                          case_priority = "elective",
                          return_index_procedures_only=TRUE,
                          verbose = TRUE
                          )

q3 <- cohort_encounter_selection(cohort_name = "Adenotonsillectomy",
                          include_OPCS =  c("F34", "F35", "E20"),
                          exclude_OPCS = c(rest_of_chapter),
                          min_age = NA,
                          max_age = NA,
                          case_priority = "elective",
                          return_index_procedures_only=TRUE,
                          verbose = TRUE
                          )

q3 <- q3 %>% filter(!EncounterKey %in% q1$EncounterKey) %>% filter(!EncounterKey %in% q2$EncounterKey)


coblation <- cohort_encounter_selection(cohort_name = "Coblation",
                          include_OPCS =  c("Y10.1", "F34.7", "F35.2"),
                          min_age = NA,
                          max_age = NA,
                         case_priority = "elective",
                         return_index_procedures_only=TRUE,
                          verbose = TRUE
                          )

electrocautery <- cohort_encounter_selection(cohort_name = "Electrocautery",
                          include_OPCS =  c("Y10.2", "Y11.1"),
                          exclude_OPCS = c("Y10.1"),
                          min_age = NA,
                          max_age = NA,
                         case_priority = "elective",
                         return_index_procedures_only=TRUE,
                          verbose = TRUE
                          )


full_cohort <- rbind(q1,q2,q3) %>% ungroup()

full_cohort <- full_cohort %>%
  mutate(coblation = ifelse(SurgicalCaseKey %in% coblation$SurgicalCaseKey,TRUE,FALSE)) %>%
  mutate(electrocautery = ifelse(SurgicalCaseKey %in% electrocautery$SurgicalCaseKey,TRUE,FALSE)) 

return(full_cohort)
}

Sensitvity Analysis

strong_opioid_list <- c("morphine sulfate", "oxycodone hydrochloride", "fentanyl", "diamorphine hydrochloride")

med_drug_names <- medication %>% filter(SimpleGenericName %in% strong_opioid_list) %>% select(GenericName, SimpleGenericName) %>% unique() %>% collect() 

med_order_drug_names <- medication_order %>% filter(SimpleGenericName %in% strong_opioid_list) %>% select(GenericName, SimpleGenericName) %>% unique() %>% collect() 

opioid_generic_names <- c(med_drug_names$GenericName, med_order_drug_names$GenericName) %>% unique()

recovery_meds <-  get_all_drug_given(encounter_keys =  df$EncounterKey,
                   drug_generic_name = opioid_generic_names,
                   search_column = "GenericName",
                  surgery_end_datetime = df$OperationInstant + df$ProcedureLengthMinutes,
                  post_op_length = 24*60,
                  timeframe = "post_op",
                  format = "all_rows",
                  variable_name = "ome_post_op",
                  verbose = FALSE) 

surg_times <- data.frame(EncounterKey =  df$EncounterKey,surgery_end_datetime = df$OperationInstant + df$ProcedureLengthMinutes)

sensititvity_df <- recovery_meds %>% select(EncounterKey, AdministrationInstant) %>% left_join(surg_times) %>% mutate(med_time = as.numeric(difftime(AdministrationInstant,surgery_end_datetime, units = "mins")))

opioid_sensititvity_df <- sensititvity_df %>% group_by(EncounterKey) %>% slice_min(med_time) %>% ungroup() %>%select( med_time) 
readmissions <-  find_readmission(index_encounter_keys = df$EncounterKey,
                 encounter_types = "ed",
                 format="all_rows",
                 variable_name = "ed_visit_in_30d",
                 cutoff = 40) 

readmissions <- readmissions %>%
  rename(ed_encounterkey = EncounterKey)%>%
  group_by(index_encounter_key) %>%
  slice_min(Date) %>%
  select(index_encounter_key, ed_encounterkey, Date)

discharge_times <- data.frame(PatientDurableKey  =  df$PatientDurableKey,
                              Index_encounter_discharge = df$EndInstant,
                              index_encounter_key = df$EncounterKey) 

readmit_sensititvity_df <- readmissions %>%
  rename(ED_visit_date = Date) %>%
  left_join(discharge_times ) %>%
  mutate(readmit_time = as.numeric(difftime(ED_visit_date,Index_encounter_discharge, units = "days")))  %>%
  filter(readmit_time >=0)%>% group_by(PatientDurableKey) %>% slice_min(readmit_time) %>% ungroup() %>% select(readmit_time)
